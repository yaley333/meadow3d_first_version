cmake_minimum_required(VERSION 3.20)
project(Meadow3D C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(glfw3 CONFIG REQUIRED)
find_package(glad CONFIG REQUIRED)
add_executable(Meadow3D main.c)
target_link_libraries(Meadow3D PRIVATE glfw glad::glad)
if(WIN32)
    target_link_libraries(Meadow3D PRIVATE opengl32)
endif()
if(MSVC)
    target_compile_options(Meadow3D PRIVATE /W4)
else()
    target_compile_options(Meadow3D PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(Meadow3D PRIVATE m)
endif()
